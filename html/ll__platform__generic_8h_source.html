749inline int ll_u8_seqlen(unsigned char b0)
750{
751 if (b0 < 0x80) return 1;
752 if ((b0 & 0xE0) == 0xC0) return 2;
753 if ((b0 & 0xF0) == 0xE0) return 3;
754 if ((b0 & 0xF8) == 0xF0) return 4;
755 return 1;
756}
757
760inline ll_codepoint_t ll_u8_decode(const char *p, int *nb)
761{
762 unsigned char b0 = (unsigned char) p[0];
763 int n = ll_u8_seqlen(b0);
764 if (nb) *nb = n;
765 if (n == 1) return (ll_codepoint_t) b0;
766 ll_codepoint_t cp = b0 & (0xFF >> (n + 1));
767 for (int i = 1; i < n; i++) {
768 unsigned char c = (unsigned char) p[i];
769 if ((c & 0xC0) != 0x80) { if (nb) *nb = 1; return (ll_codepoint_t) b0; }
770 cp = (cp << 6) | (c & 0x3F);
771 }
772 return cp;
773}
774
776inline int ll_u8_encode(ll_codepoint_t cp, char *out)
777{
778 if (cp < 0x80) { out[0] = (char) cp; return 1; }
779 if (cp < 0x800) { out[0] = (char)(0xC0 | (cp >> 6)); out[1] = (char)(0x80 | (cp & 0x3F)); return 2; }
780 if (cp < 0x10000) { out[0] = (char)(0xE0 | (cp >> 12)); out[1] = (char)(0x80 | ((cp >> 6) & 0x3F));
781 out[2] = (char)(0x80 | (cp & 0x3F)); return 3; }
782 out[0] = (char)(0xF0 | (cp >> 18)); out[1] = (char)(0x80 | ((cp >> 12) & 0x3F));
783 out[2] = (char)(0x80 | ((cp >> 6) & 0x3F)); out[3] = (char)(0x80 | (cp & 0x3F)); return 4;
784}
785
787inline LL_int32 ll_u8_strlen(const char *s)
788{
789 LL_int32 n = 0;
790 for (const char *p = s; *p; n++) p += ll_u8_seqlen((unsigned char) *p);
791 return n;
792}

796inline const char *ll_u8_index(const char *s, LL_int32 k)
797{
798 const char *p = s;
799 for (LL_int32 i = 0; i < k; i++) {
800 if (!*p) return 0;
801 p += ll_u8_seqlen((unsigned char) *p);
802 }
803 return p;
804}

817//The AsciiConverter class mitigates printf formatting problems caused by different int sizes on different processors, when using printf-style format string (%u, %d %lu %ld etc).
818class AsciiConverter {
819public:
820
821 char *dec(Word_t n) {
822 LL_int32 ix = buffer_size - 1;
823 char *ptr = &(buffer[ix]);
824
825 *ptr = '\0';
826 do {
827 Word_t d = n % 10;
828 n /= 10;
829 char c = d + '0';
830 *--ptr = c;
831 } while (n);
832 return ptr;
833 }
834
835 char *dec(LL_int32 n) {
836 if (n >= 0) return dec((Word_t) n);
837 if (n == INT32_MIN) return dec((LL_int64) n);
838 char *ptr = dec(-n);
839 *--ptr = '-';
840 return ptr;
841 }
842
857 template <typename T>
858 typename std::enable_if<std::is_same<T, int>::value && !std::is_same<LL_int32, int>::value,
859 char *>::type
860 dec(T n) { return dec((LL_int32) n); }
861
862 char *dec(LL_int64 n) {
863 // Full 64-bit signed decimal; avoids conflict with dec(Word_t) on 64-bit platforms.
864 LL_int32 ix = buffer_size - 1;
865 char *ptr = &(buffer[ix]);
866 *ptr = '\0';
867 bool neg = (n < 0);
868 uint64_t uv = neg ? (n == INT64_MIN ? (uint64_t) 9223372036854775808ULL : (uint64_t) -n)
869 : (uint64_t) n;
870 do { *--ptr = (char)('0' + (uv % 10)); uv /= 10; } while (uv);
871 if (neg) *--ptr = '-';
872 return ptr;
873 }
874
875 char *hex(Word_t n) {
876 LL_int32 ix = buffer_size - 1;
877 char *ptr = &(buffer[ix]);
878 LL_int32 nibs = sizeof(Word_t) * 2;
879
880 *ptr = '\0';
881 while (nibs--) {
882 Word_t d = n & 0x0f;
883 n >>= 4;
884 char c = (d < 10) ? (d + '0') : (d - 10 + 'a');
885 *--ptr = c;
886 }
887 return ptr;
888
889 }
890
891 char *dec(LL_float32 n) {
892 snprintf(buffer, buffer_size, "%f", (LL_float64) n);
893 return buffer;
894 }
895
896private:
897 static const LL_int32 buffer_size = 128;
898 char buffer[buffer_size];
899};
